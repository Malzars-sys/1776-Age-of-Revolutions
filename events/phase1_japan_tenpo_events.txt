namespace = phase1_japan_tenpo

# 1776-compatible lead-in to the vanilla Tenpo Crisis.
phase1_japan_tenpo.1 = {
	type = country_event
	placement = root

	duration = 3

	title = phase1_japan_tenpo.1.t
	desc = phase1_japan_tenpo.1.d
	flavor = phase1_japan_tenpo.1.f

	event_image = {
		video = "asia_dead_cattle_poor_harvest"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		c:JAP ?= this
		game_date >= 1787.1.1
		game_date < 1791.1.1
		NOT = { has_variable = phase1_japan_tenmei_lessons_done }
	}

	immediate = {
		set_variable = phase1_japan_tenmei_lessons_done
	}

	option = {
		name = phase1_japan_tenpo.1.a
		default_option = yes
		add_loyalists = {
			value = small_radicals
			strata = lower
		}
	}

	option = {
		name = phase1_japan_tenpo.1.b
		add_loyalists = {
			value = small_radicals
			strata = lower
		}
		add_radicals = {
			value = small_radicals
			strata = upper
		}
	}

	option = {
		name = phase1_japan_tenpo.1.c
		add_radicals = {
			value = small_radicals
			strata = lower
		}
	}
}

phase1_japan_tenpo.6 = {
	type = country_event
	placement = root

	duration = 3

	title = phase1_japan_tenpo.6.t
	desc = phase1_japan_tenpo.6.d
	flavor = phase1_japan_tenpo.6.f

	event_image = {
		video = "asia_dead_cattle_poor_harvest"
	}

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	trigger = {
		c:JAP ?= this
		has_dlc_feature = ep2_content
		game_date >= 1833.1.1
		game_date < 1836.1.1
		has_variable = phase1_japan_bad_harvests_1832
		has_variable = phase1_japan_rice_price_crisis_done
		NOT = { has_variable = phase1_japan_tenpo_crisis_started_from_1776_mod }
		NOT = { has_journal_entry = je_tenpo_crisis }
	}

	immediate = {
		set_variable = phase1_japan_tenpo_crisis_started_from_1776_mod
		set_variable = {
			name = tenpo_gdp_goal
			value = {
				value = root.gdp
				multiply = 1.35
			}
		}
	}

	option = {
		name = phase1_japan_tenpo.6.a
		default_option = yes
		add_journal_entry = {
			type = je_tenpo_crisis
		}
		trigger_event = {
			id = tenpo_events.1
			days = 1
			popup = yes
		}
	}
}
