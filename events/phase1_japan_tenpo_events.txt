namespace = phase1_japan_tenpo

# 1776-compatible lead-in to the vanilla Tenpo Crisis.
phase1_japan_tenpo.1 = {
	type = country_event
	placement = root

	duration = 3

	title = phase1_japan_tenpo.1.t
	desc = phase1_japan_tenpo.1.d
	flavor = phase1_japan_tenpo.1.f

	event_image = {
		video = "asia_dead_cattle_poor_harvest"
	}

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		c:JAP ?= this
		game_date >= 1787.1.1
		game_date < 1791.1.1
		NOT = { has_variable = phase1_japan_tenmei_lessons_done }
	}

	immediate = {
		set_variable = phase1_japan_tenmei_lessons_done
	}

	option = {
		name = phase1_japan_tenpo.1.a
		default_option = yes
		add_treasury = { value = root.gdp multiply = -0.005 }
		custom_tooltip = aor1776_japan_granary_policy_tt
		set_variable = { name = aor1776_japan_grain_procurement_active days = 730 }
		set_variable = aor1776_japan_grain_reserves
		every_scope_state = {
		  limit = { any_scope_building = { OR = {
		    is_building_type = building_rice_farm
		    is_building_type = building_wheat_farm
		    is_building_type = building_rye_farm
		    is_building_type = building_maize_farm
		    is_building_type = building_millet_farm
		    is_building_group = bg_subsistence_agriculture
		  } } }
		  add_modifier = { name = aor1776_japan_grain_procurement days = 730 }
		}
		every_scope_state = { add_modifier = { name = aor1776_japan_granary_network } }
		add_radicals = { value = small_radicals pop_type = peasants }
		ig:ig_rural_folk ?= { add_modifier = { name = aor1776_japan_ig_opposition days = 1825 } }
		ig:ig_landowners ?= { add_modifier = { name = aor1776_japan_ig_support days = 1825 } }
	}

	option = {
		name = phase1_japan_tenpo.1.b
		add_treasury = { value = root.gdp multiply = -0.003 }
		every_scope_state = { add_modifier = { name = aor1776_japan_village_relief days = 1095 } }
		add_loyalists = { value = small_radicals strata = lower }
		ig:ig_rural_folk ?= { add_modifier = { name = aor1776_japan_ig_support days = 1095 } }
	}

	option = {
		name = phase1_japan_tenpo.1.c
		add_modifier = { name = aor1776_japan_government_inaction days = 1825 }
		ig:ig_landowners ?= { add_modifier = { name = aor1776_japan_ig_support days = 1825 } }
		add_radicals = { value = small_radicals strata = lower }
	}
}

phase1_japan_tenpo.6 = {

	flavor = phase1_japan_tenpo.6.f

	event_image = {
		video = "asia_dead_cattle_poor_harvest"
	}

	icon = "gfx/interface/icons/event_icons/event_protest.dds"

	trigger = {
		c:JAP ?= this
		has_dlc_feature = ep2_content
		game_date >= 1833.1.1
		game_date < 1836.1.1
		has_variable = phase1_japan_bad_harvests_1832
		has_variable = phase1_japan_rice_price_crisis_done
		NOT = { has_variable = phase1_japan_tenpo_crisis_started_from_1776_mod }
		NOT = { has_journal_entry = je_tenpo_crisis }
	}

	immediate = {
		set_variable = phase1_japan_tenpo_crisis_started_from_1776_mod
		set_variable = {
			name = tenpo_gdp_goal
			value = {
				value = root.gdp
				multiply = 1.35
			}
		}
	}

	option = {
		name = phase1_japan_tenpo.6.a
		default_option = yes
		add_journal_entry = {
			type = je_tenpo_crisis
		}
		trigger_event = {
			id = tenpo_events.1
			days = 1
			popup = yes
		}
	}
}
