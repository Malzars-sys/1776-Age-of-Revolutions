# A non-marginal Trade Unions IG may demand compensation while the quota trial
# is active. The immediate flag keeps the offer unique even if the player declines.
birth_quota_events.15 = {
	type = country_event
	placement = root
	title = birth_quota_events.15.t
	desc = birth_quota_events.15.d
	flavor = birth_quota_events.15.f
	duration = 3
	event_image = { video = "birth_quota_scene" }
	icon = "gfx/interface/icons/journal_entry_icons/birth_quota_trial.dds"
	trigger = {
		has_law = law_type:law_birth_quotas
		has_journal_entry = je_birth_quota_trial
		NOT = { has_variable = birth_quota_housewife_wage_decided }
		ig:ig_trade_unions ?= { ig_counts_as_marginal = no }
		active_law:lawgroup_rights_of_women ?= {
			NOT = { has_amendment = amendment_type:amendment_birth_quota_housewife_wage }
		}
	}
	immediate = { set_variable = birth_quota_housewife_wage_decided }
	option = {
		name = birth_quota_events.15.a
		default_option = yes
		active_law:lawgroup_rights_of_women = {
			add_amendment = {
				type = amendment_birth_quota_housewife_wage
				sponsor = PREV.ig:ig_trade_unions
			}
		}
	}
	option = {
		name = birth_quota_events.15.b
	}
}
