french_revolution_mod_events.5 = { # Rise of Radicalism
	type = country_event
	placement = root
	title = french_revolution_mod_events.5.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = 1848.4.d
				trigger = {
					any_interest_group = {
						OR = {
							is_powerful = yes
							is_insurrectionary = yes
						}
						leader = {
							has_ideology = ideology:ideology_radical
						}
					}
				}
			}
			triggered_desc = {
				desc = 1848.4.d_no_leader
				trigger = {
					NOT = {
						any_interest_group = {
							OR = {
								is_powerful = yes
								is_insurrectionary = yes
							}
							leader = {
								has_ideology = ideology:ideology_radical
							}
						}
					}
				}
			}
		}
	}
	flavor = french_revolution_mod_events.5.f

	duration = 3

	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	trigger = {
		# triggered by journal entry start
	}

	immediate = {
		random_interest_group = {
			limit = {
				OR = {
					is_powerful = yes
					is_insurrectionary = yes
				}
				leader = {
					has_ideology = ideology:ideology_radical
				}
			}
			leader = {
				save_scope_as = radical_fra_leader_scope
			}
		}
	}

	cancellation_trigger = {
		NOT = {
			has_journal_entry = je_fr_rev_summon_estates
		}
	}

	option = { # reactionary option
		name = 1848.4.a
		default_option = yes
		scope:radical_leader_scope ?= {
			add_modifier = {
				name = 1848_popular_radical
			}
		}
		show_as_tooltip = {
			if = {
				limit = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_radical
						}
					}
				}
				create_political_movement = { type = movement_radical }
			}
		}
#		scope:radical_fra_movement_scope = {
#			add_modifier = {
#				name = suppressed_rally_ig
#				days = normal_modifier_time
#			}
#		}
		add_modifier = {
			name = suppressing_radicals
			days = short_modifier_time
		}
		add_loyalists = {
			strata = upper
			value = very_large_radicals
		}
		add_radicals = {
			value = small_radicals
			strata = middle
		}
		add_radicals = {
			value = small_radicals
			strata = lower
		}
		ai_chance = {
			base = 66
		}
	}
	option = { # radical option
		name = 1848.4.b
		custom_tooltip = {
			text = fr_rev_summon_estates_advance_30_tt
			je:je_fr_rev_summon_estates ?= {
				add_progress = { value = 30 name = je_estates_reform_bar }
			}
		}
		scope:radical_leader_scope ?= {
			add_modifier = {
				name = 1848_popular_radical
			}
		}
		if = {
			limit = {
				any_interest_group = {
					ig_counts_as_marginal = no
					can_have_radical_leader = yes
					is_in_government = no
				}
			}
			random_interest_group = {
				limit = {
					ig_counts_as_marginal = no
					can_have_radical_leader = yes
					is_in_government = no
				}
				leader = {
					set_ideology = ideology:ideology_radical
					add_modifier = {
						name = 1848_popular_radical
					}
				}
			}
		}
		show_as_tooltip = {
			if = {
				limit = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_radical
						}
					}
				}
				create_political_movement = { type = movement_radical }
			}
		}
		add_radicals = {
			value = very_large_radicals
			strata = middle
		}
		add_radicals = {
			value = very_large_radicals
			strata = lower
		}
		add_radicals = {
			value = medium_radicals
		}
		ai_chance = {
			base = 33
		}
	}
}
