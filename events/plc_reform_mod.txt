	duration = 1

	cooldown = { days = very_long_modifier_time }
	

	trigger = {
		# Triggered by JE
	}

	immediate = {

	}
	
	option = { # Loosen purse strings
		name = plc_reform_modevents.3.a
		default_option = yes
		ai_chance = {
			base = 25
			modifier = {
				trigger = {
					OR = {
						tax_level < high
						net_total_income > money_amount_multiplier_medium
					}
				}
				add = 35
			}
			modifier = { #don't do if broke
				trigger = {
					AND = {
						net_total_income < 0
						tax_level >= high
					}
				}
				add = -50
			}
		}
        add_modifier = {
			name = mod_farming_subsidies
            days = normal_modifier_time
        }
	}
	option = { # tell them go away
		name = plc_reform_modevents.3.b
		ai_chance = {
			base = 10 # this is the default pick for the AI
		}
		ig:ig_rural_folk = {
			add_modifier = {
				name = mod_go_away_farmer
			}
		}
	}
	
}
