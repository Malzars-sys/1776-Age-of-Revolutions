law_birth_quotas = {
	group = lawgroup_rights_of_women
	parent = law_no_womens_rights # Existing patriarchal and traditionalist stances carry over.
	icon = "gfx/interface/icons/law_icons/women_in_the_fields.dds"
	progressiveness = -50

	unlocking_technologies = {
		mass_propaganda
	}

	modifier = {
		state_birth_rate_mult = 0.15
		state_working_adult_ratio_add = -0.15
	}

	on_enact = {
		recalculate_pop_ig_support = yes
		trigger_event = { id = birth_quota_events.4 days = 1 }
		save_scope_as = quota_enacting_country
		every_country = {
			limit = {
				NOT = { this = scope:quota_enacting_country }
				is_country_type = recognized
				has_diplomatic_relevance = scope:quota_enacting_country
			}
			if = {
				limit = {
					OR = {
						ruler = { has_ideology = ideology:ideology_feminist }
						ruler = { has_ideology = ideology:ideology_humanitarian }
						ruler = { has_ideology = ideology:ideology_radical }
						any_interest_group = {
							is_in_government = yes
							has_ideology = ideology:ideology_feminist_ig
						}
					}
				}
				trigger_event = { id = birth_quota_events.8 days = 7 }
			}
			else_if = {
				limit = {
					OR = {
						ruler = { has_ideology = ideology:ideology_traditionalist }
						ruler = { has_ideology = ideology:ideology_fascist }
						any_interest_group = {
							is_in_government = yes
							has_ideology = ideology:ideology_patriarchal
						}
					}
				}
				trigger_event = { id = birth_quota_events.9 days = 7 }
			}
			else = {
				trigger_event = { id = birth_quota_events.10 days = 7 }
			}
		}
	}

	ai_will_do = {
		OR = {
			ruler = { has_ideology = ideology:ideology_traditionalist }
			ruler = { has_ideology = ideology:ideology_fascist }
			any_political_movement = {
				is_political_movement_type = movement_natalist
			}
		}
	}
}
