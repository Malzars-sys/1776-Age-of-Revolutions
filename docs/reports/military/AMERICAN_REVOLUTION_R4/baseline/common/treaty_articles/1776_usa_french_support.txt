# R3: advances are paired treasury payments tracked by the debt pact. There is
# deliberately no is_money_transfer flag here: that would pay a second time.
# Legacy R2 money clauses are invalidated, never turned into retrospective debt.
aor1776_support_money = {
 kind = directed cost = 0
 icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/bankroll_treaties.dds"
 flags = { is_money_transfer }
 maintenance_paid_by = source_country
 required_inputs = { quantity }
 visible = { always = no }
 possible = { always = no }
 requirement_to_maintain = { trigger = { always = no } }
 quantity_min_value = { value = 1 }
 quantity_max_value = { value = 21000 }
}

aor1776_support_supplies = {
 kind = directed cost = 0
 icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/goods_transfer_treaties.dds"
 flags = { is_goods_transfer }
 maintenance_paid_by = source_country
 required_inputs = { quantity goods }
 visible = { OR = { c:FRA ?= this c:USA ?= this } }
 possible = {
  OR = { c:FRA ?= this c:USA ?= this }
  is_market_reachable_for_trade = scope:other_country.market
 }
 can_ratify = {
  scope:source_country = { c:FRA ?= this }
  scope:target_country = { c:USA ?= this }
 }
 requirement_to_maintain = { trigger = { aor1776_french_support_relationship_valid = yes
   c:USA = { has_variable = aor1776_support_active } } }
 non_fulfillment = {
  consequences = freeze
  conditions = {
   weekly = {
    scope:article = {
     OR = {
      source_country = { supply_network_strength < 0.5 }
      NOT = { source_country = { is_market_reachable_for_trade = scope:article.target_country.market } }
      "scope:article.input_market_goods(scope:article.source_country)" = { treaty_exports > market_export_cap }
     }
    }
   }
  }
 }
 goods_valid_trigger = { scope:input = { OR = { this = g:small_arms this = g:artillery } } }
 quantity_min_value = { value = 1 }
 quantity_max_value = { value = 12 }
}

aor1776_deferred_debt_article = {
 kind = directed cost = 0
 icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/bankroll_treaties.dds"
 maintenance_paid_by = source_country
 usage_limit = once_per_side
 visible = { NOT = { is_country_type = decentralized } }
 possible = { NOT = { has_war_with = scope:other_country } }
 can_ratify = {
  scope:source_country = { gold_reserves >= 10000 in_default = no }
  scope:target_country = {
   OR = {
    AND = { c:USA ?= this has_variable = aor1776_support_offer NOT = { has_variable = aor1776_debt_creditor } scope:source_country = { c:FRA ?= this } }
    AND = { has_variable = aor1776_debt_creditor var:aor1776_debt_creditor = scope:source_country }
   }
  }
 }
 requirement_to_maintain = {
  trigger = {
   scope:target_country = {
    OR = {
     NOT = { has_variable = aor1776_debt_revolution_credit }
     game_date < 1783.9.4
    }
    OR = {
     has_variable = aor1776_support_offer
     AND = {
      has_variable = aor1776_debt_creditor
      var:aor1776_debt_creditor = scope:source_country
     }
    }
   }
  }
 }
 on_entry_into_force = {
  scope:article_options.target_country = {
   aor1776_debt_open_effect = { CREDITOR = scope:article_options.source_country }
  }
 }
 # Withdrawal ends the treaty clause, but never silently erases its debt pact.
}
