   triggered_desc = { trigger = { c:USA = { var:aor1776_support_offered_interest = 5 } } desc = aor1776_american_diplomacy.4.t_high }
   triggered_desc = { trigger = { always = yes } desc = aor1776_american_diplomacy.4.t }
  }
 }
 desc = {
  first_valid = {
   triggered_desc = { desc = aor1776_american_diplomacy.4.d_high trigger = { c:USA = { var:aor1776_support_offered_interest = 5 } } }
   triggered_desc = { trigger = { always = yes } desc = aor1776_american_diplomacy.4.d }
  }
 }
 flavor = aor1776_american_diplomacy.4.f
 event_image = { video = "ep2_intellectual_writer" }
 icon = "gfx/interface/icons/event_icons/waving_flag.dds"
 duration = 30
 trigger = { c:USA ?= this has_variable = aor1776_support_offer }
 immediate = {
  # Recover terms for an offer scheduled by R4 before this correction loaded.
  if = {
   limit = { NOT = { has_variable = aor1776_support_offered_interest } }
   set_variable = { name = aor1776_support_offered_interest value = 0 days = 45 }
   if = {
    limit = { var:aor1776_support_offer_level = 2 }
    set_variable = { name = aor1776_support_offered_interest value = 5 days = 45 }
   }
  }
 }
 option = {
  name = aor1776_american_diplomacy.4.a default_option = yes
  ai_chance = { base = 95 }
  trigger = { aor1776_french_support_can_accept = yes var:aor1776_support_offered_interest = 0 }
  custom_tooltip = aor1776_r5_ratification_no_interest_tt
  hidden_effect = { aor1776_support_accept_effect = yes }
 }
 option = {
  name = aor1776_american_diplomacy.4.c default_option = yes

 }
 option = {
  name = aor1776_american_diplomacy.4.b
  ai_chance = { base = 5 }
  aor1776_support_stop_effect = yes
  set_variable = { name = aor1776_support_cooldown days = 90 }
 }
}
