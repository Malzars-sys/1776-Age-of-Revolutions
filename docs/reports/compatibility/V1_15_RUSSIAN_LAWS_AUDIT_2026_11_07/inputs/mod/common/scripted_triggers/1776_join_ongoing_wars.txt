# Adapted from Join Ongoing Wars (The Lizerd King); native 1.13 war scopes.
aor1776_can_intervene_trigger = {
 $ACTOR$ = {
  NOT = { is_country_type = decentralized }
  is_revolutionary = no is_secessionist = no
  aggressive_diplomatic_plays_permitted = yes
  NOT = { has_war_with = $FRIEND$ }
  NOT = { has_truce_with = $FRIEND$ }
 }
 $FRIEND$ = {
  NOT = { is_country_type = decentralized }
  is_at_war = yes
  OR = {
   aggressive_diplomatic_plays_permitted = yes
   is_subject_of = $ACTOR$
   AND = { c:USA ?= this aor1776_usa_independence_war_active = yes }
  }
  OR = {
   has_strategic_adjacency = $ACTOR$
   AND = { any_scope_state = { is_coastal = yes } $ACTOR$ = { any_scope_state = { is_coastal = yes } } }
  }
  any_scope_war = {
   OR = {
    is_warleader = $FRIEND$
    $ACTOR$ = { any_scope_treaty = { binds = $FRIEND$ any_scope_article = { has_type = alliance } } }
   }
   NOT = { is_war_participant = $ACTOR$ }
   aor1776_intervention_stake_in_war = { ACTOR = $ACTOR$ FRIEND = $FRIEND$ }
  }
 }
 NOT = { any_country = { has_war_with = $FRIEND$ has_truce_with = $ACTOR$ } }
}

# War scope. Check each war separately, so relevance to one conflict does not
# grant permission to join every other war fought by the supported country.
aor1776_intervention_stake_in_war = {
 custom_tooltip = {
  text = aor1776_r5_american_intervention_stake_tt
  OR = {
   NOT = { AND = { is_war_participant = c:USA is_war_participant = c:GBR } }
   $ACTOR$ = {
    OR = {
     has_interest_marker_in_region = sr:region_atlantic_coast
     AND = {
      OR = { c:FRA ?= this c:SPA ?= this }
      $FRIEND$ = { has_war_with = c:GBR }
     }
     AND = {
      c:POR ?= this
      any_scope_treaty = { binds = c:GBR any_scope_article = { has_type = alliance } }
      $FRIEND$ = { OR = { c:GBR ?= this is_in_war_together = c:GBR } }
     }
     AND = {
      has_diplomatic_pact = { who = c:GBR type = rivalry }
      $FRIEND$ = { has_war_with = c:GBR }
     }
    }
   }
  }
 }
}
aor1776_revolution_intervention_eligible = {
 aor1776_usa_independence_war_active = yes
 NOT = { has_war_with = c:GBR }
 aor1776_can_intervene_trigger = { ACTOR = $ACTOR$ FRIEND = c:USA }
}
