# State scope. All automatic intervention is AI-only.
aor1776_ai_regional_infrastructure_shortage = {
	owner = { is_ai = yes }
	any_scope_building = {
		is_building_type = building_railway
		building_has_goods_shortage = yes
	}
}

aor1776_ai_supply_aluminium = {
	sg:aluminium = {
		state_goods_has_local_goods_shortage = no
		state_goods_pricier <= 0.25
	}
	OR = {
		AND = {
		market_access >= 0.95
		is_isolated_from_market = no
		market = { mg:aluminium = {
			market_goods_has_goods_shortage = no
			market_goods_sell_orders > 0
			market_goods_sell_orders >= aor1776_ai_market_reserve_aluminium
		} }
		}
		sg:aluminium = {
			state_goods_production > 0
			state_goods_production >= aor1776_ai_local_reserve_aluminium
		}
	}
}

aor1776_ai_prioritize_aluminium = {
	owner = { is_ai = yes }
	OR = {
		has_variable = aor1776_ai_need_aluminium
		AND = {
			market_access >= 0.95
			is_isolated_from_market = no
			owner = { has_variable = aor1776_ai_connected_need_aluminium }
		}
	}
}
