# AI-only sustainable maintenance. No construction, resources or subsidies.
# Generated from recipes by tools/regional_infrastructure_shortage_patch.py.
aor1776_ai_apply_regional_infrastructure_pm = {
	if = {
		limit = { NOT = { any_scope_building = {
			is_building_type = building_railway
			has_active_production_method = $PM$
		} } }
		activate_production_method = { building_type = building_railway production_method = $PM$ }
		set_variable = aor1776_ai_regional_changed_this_pulse
		set_variable = { name = aor1776_ai_regional_cooldown days = 180 }
	}
}

# Rank all states, avoiding out-of-range positions in small countries.
aor1776_ai_mark_canal_gdp_position = {
	if = {
		limit = { num_states > $POSITION$ }
		ordered_scope_state = {
			order_by = gdp
			position = $POSITION$
			check_range_bounds = no
			set_variable = aor1776_ai_top_gdp_canal_state
		}
	}
}

aor1776_ai_choose_regional_roads = {
	if = {
		limit = { aor1776_ai_sustainable_pm_paved_road_network = yes }
		aor1776_ai_apply_regional_infrastructure_pm = { PM = pm_paved_road_network }

aor1776_ai_select_sustainable_regional_infrastructure = {
	aor1776_ai_choose_regional_roads = yes
	aor1776_ai_choose_regional_rail = yes
	if = {
		limit = { any_scope_building = {
			is_building_type = building_railway
			has_active_production_method = pm_no_rail_network
		} }
		aor1776_ai_apply_regional_infrastructure_pm = { PM = pm_no_passenger_trains }
	}
	else = { aor1776_ai_choose_regional_wagons = yes }
	if = {
		limit = { has_variable = aor1776_ai_top_gdp_canal_state }
		aor1776_ai_choose_regional_canals = yes
	}
	else = { aor1776_ai_apply_regional_infrastructure_pm = { PM = pm_no_canal_network } }
}

# Two consecutive observed shortages; not an estimate of engine penalty.
# Healthy upgrades wait 180 days after any actual PM change.
aor1776_ai_refresh_regional_infrastructure = {
	if = {
		limit = { is_ai = yes any_scope_building = { is_building_type = building_railway } }
		set_variable = { name = aor1776_ai_reserved_aluminium value = 0 }
		set_variable = { name = aor1776_ai_reserved_cement value = 0 }
		set_variable = { name = aor1776_ai_reserved_coal value = 0 }
		set_variable = { name = aor1776_ai_reserved_electricity value = 0 }
		set_variable = { name = aor1776_ai_reserved_engines value = 0 }
		set_variable = { name = aor1776_ai_reserved_heavy_petroleum_products value = 0 }
		set_variable = { name = aor1776_ai_reserved_iron value = 0 }
		set_variable = { name = aor1776_ai_reserved_limestone value = 0 }
		set_variable = { name = aor1776_ai_reserved_refined_fuels value = 0 }
		set_variable = { name = aor1776_ai_reserved_steel value = 0 }
		set_variable = { name = aor1776_ai_reserved_tools value = 0 }
		set_variable = { name = aor1776_ai_reserved_wood value = 0 }
		every_scope_state = {
			limit = { has_variable = aor1776_ai_top_gdp_canal_state }
			remove_variable = aor1776_ai_top_gdp_canal_state
		}
		aor1776_ai_mark_canal_gdp_position = { POSITION = 0 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 1 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 2 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 3 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 4 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 5 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 6 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 7 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 8 }
		aor1776_ai_mark_canal_gdp_position = { POSITION = 9 }
		every_scope_state = {
			limit = { has_building = building_railway }
			remove_variable = aor1776_ai_regional_changed_this_pulse
			save_scope_as = aor1776_ai_regional_state
			set_variable = { name = aor1776_ai_regional_levels value = b:building_railway.level }
			aor1776_ai_record_regional_supply_needs = yes
			# Removing canals outside the top ten is safe and must not wait
			# for an upgrade cooldown; it cannot increase input demand.
			if = {
				limit = { NOT = { has_variable = aor1776_ai_top_gdp_canal_state } }
				aor1776_ai_apply_regional_infrastructure_pm = { PM = pm_no_canal_network }
			}
			if = {
				limit = { aor1776_ai_regional_infrastructure_shortage = yes }
				if = {
					limit = { has_variable = aor1776_ai_regional_shortage_previous }
					aor1776_ai_select_sustainable_regional_infrastructure = yes
				}
				set_variable = { name = aor1776_ai_regional_shortage_previous days = 45 }
			}
			else = {
				remove_variable = aor1776_ai_regional_shortage_previous
				if = {
					limit = { NOT = { has_variable = aor1776_ai_regional_cooldown } }
					aor1776_ai_select_sustainable_regional_infrastructure = yes
				}
			}
			if = {
				limit = { has_variable = aor1776_ai_regional_changed_this_pulse }
				aor1776_ai_reserve_regional_demand = yes
			}
		}
	}
}
