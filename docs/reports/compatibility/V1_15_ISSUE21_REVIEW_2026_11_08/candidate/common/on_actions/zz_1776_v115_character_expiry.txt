# Additive child action; leaves the existing monthly character pulse intact.
on_monthly_pulse_character = { on_actions = { aor1776_v115_event_immortality_expiry } }
aor1776_v115_event_immortality_expiry = {
	effect = {
		if = { # Temporary Immortality
			limit = {
				is_immortal = yes
				has_variable = temporary_immortality_var
				var:temporary_immortality_var != 0
			}
			change_variable = { name = temporary_immortality_var subtract = 1 }
		}
		else_if = {
			limit = {
				is_immortal = yes
				has_variable = temporary_immortality_var
				var:temporary_immortality_var = 0
			}
			set_character_immortal = no
		}
	}
}
