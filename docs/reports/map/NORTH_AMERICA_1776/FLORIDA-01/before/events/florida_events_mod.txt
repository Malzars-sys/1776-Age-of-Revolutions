# An Offer for florida
florida.1 = {
	type = country_event
	placement = ROOT

	title = florida.1.t
	desc = florida.1.d
	flavor = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:florida_seller = {
						c:SPA ?= this
					}
				}
				desc = florida.1.f_spa
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = florida.1.f
			}
		}
	}

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	trigger = {
	}

	immediate = {
		# Scopes set in florida.4
	}

	option = {
		name = florida.1.a
		add_modifier = {
			name = florida_spa_big_money 
			months = 12
			is_decaying = no
		}
		scope:florida_buyer = {
			trigger_event = { id = florida.2 days = 0 popup = yes }
		}
		show_as_tooltip = {
			random_state = {
				limit = {
					state_region = s:STATE_FLORIDA
					owner = {
						OR = {
							this = scope:florida_seller
							is_subject_of = scope:florida_seller
						}
					}
				}
				state_region = {
					remove_claim = root
					add_claim = scope:florida_buyer
				}
				set_state_owner = scope:florida_buyer
			}
			change_relations = {
				country = scope:florida_buyer
				value = 10
			}
		}
		ai_chance = {
			base = 75
		}
	}
	option = {
		name = florida.1.b
		default_option = yes
		scope:florida_buyer = {
			trigger_event = { id = florida.3 days = 0 popup = yes }
		}
		show_as_tooltip = {
			change_relations = {
				country = scope:florida_buyer
				value = -20
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					scope:florida_seller = {
						NOT = {
							has_variable = florida_sale_cooldown #Unsolicited offer
						}
					}
				}
				add = 25
			}
		}
	}
}

# Making an Offer
florida.4 = {
	type = country_event
	placement = ROOT

	title = florida.4.t
	desc = florida.4.d
	flavor = florida.4.f

	duration = 3

	event_image = {
		video = "unspecific_signed_contract"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	trigger = {
	}

	immediate = {
		root = {
			save_scope_as = florida_buyer
		}
		random_state = {
			limit = {
				state_region = s:STATE_FLORIDA
				owner = {
					NOT = {
						is_country_type = decentralized
					}
				}
			}
			save_scope_as = florida_scope
			if = { 
				limit = {
					owner = {
						is_subject = yes
						capital = {
							state_region = s:STATE_FLORIDA
						}
					}
				}
				owner.top_overlord = {
					save_scope_as = florida_seller
				}
			}
			else = { # For other cases
				owner = {
					save_scope_as = florida_seller
				}
			}
		}
	}

	option = {
		name = florida.4.a
		ai_chance = {
			base = 1
		}
		scope:florida_seller = {
			trigger_event = {
				id = florida.1
				days = 0
				popup = yes
			}
		}
	}
	option = { # perhaps another time
		name = florida.4.c
		default_option = yes
		ai_chance = {
			base = 0
		}
		scope:florida_seller = {
			post_notification = florida_negotiations_fail
		}
	}
}
