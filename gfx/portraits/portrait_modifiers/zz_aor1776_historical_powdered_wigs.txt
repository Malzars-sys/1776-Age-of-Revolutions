	usage = game
	selection_behavior = weighted_random

				range = { 0.5 0.5 }
			}
		}
		weight = {
			base = 0
			modifier = {
				add = 90000
				scope:character ?= {
					has_variable = aor1776_historical_powdered_wig
					is_historical = yes
					is_female = no
					is_adult = yes
				}
			}
		}
	}
}
