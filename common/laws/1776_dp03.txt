# Bounded starting abstraction retaining the existing law effects, not a claim of modern constitutional institutions.
law_aor1776_dp03_customary_bondage = {
	group = lawgroup_slavery

	progressiveness = -50

	icon = "gfx/interface/icons/law_icons/debt_slavery.dds"

	unlocking_technologies = { international_relations }

	disallowing_laws = {
		law_multicultural
		law_affirmative_action
	}

	on_activate = {
		set_variable = {
			name = slavery_recently_enacted
			value = yes
			days = 1825
		}
		every_scope_state = {
			limit = {
				NOT = {
					any_scope_pop = {
						is_pop_type = slaves
					}
				}
			}
			set_variable = {
				name = former_free_state
				value = yes
				days = 1825
			}
		}
		ig:ig_landowners ?= {
			switch_from_colonial_slavery_to_pro_slavery = yes
		}

		hidden_effect = {
			if = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_pro_slavery
						has_core_ideology = ideology:ideology_pro_slavery_colonial_movement
					}
				}
				random_political_movement = {
					limit = {
						is_political_movement_type = movement_pro_slavery
						has_core_ideology = ideology:ideology_pro_slavery_colonial_movement
					}
					set_core_ideology = ideology_pro_slavery_movement
				}
			}
		}
	}

	modifier = {
		interest_group_ig_landowners_pol_str_mult = 0.25
	}

	# AI should never enact debt slavery
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

 parent = law_debt_slavery
 is_visible = { OR = { c:D3G ?= this c:D3S ?= this } }
 can_enact = { OR = { c:D3G ?= this c:D3S ?= this } }
}

# Bounded starting abstraction retaining the existing law effects, not a claim of modern constitutional institutions.
law_aor1776_dp03_toleration = {
	group = lawgroup_church_and_state

	icon = "gfx/interface/icons/law_icons/freedom_of_conscience.dds"

	progressiveness = 0

	unlocking_technologies = { international_relations }

	disallowing_laws = {
		law_theocracy
	}

	on_activate = {
		if = {
			limit = {
				has_law_or_variant = law_type:law_state_atheism
			}
			set_state_religion = root.country_definition.religion
		}
		if = {
			limit = {
				has_modifier = antireligious_policies
			}
			remove_modifier = antireligious_policies
		}
		recalculate_pop_ig_support = yes
	}

	modifier = {
		interest_group_ig_devout_pol_str_mult = 0.15
		country_authority_add = 100
		country_state_religion_wages_mult = 0.05
		country_non_state_religion_wages_mult = -0.05

		country_acceptance_state_religion_add = 20
		country_acceptance_shared_religious_trait_add = 15
		country_acceptance_shared_religious_trait_group_add = 5
		country_acceptance_no_shared_religious_trait_add = 0

		country_allow_conversion_violent_hostility_bool = yes
		country_allow_conversion_cultural_erasure_bool = yes
		country_allow_conversion_open_prejudice_bool = yes
		country_allow_conversion_second_rate_citizen_bool = no
		country_allow_conversion_full_acceptance_bool = no
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_church_and_state_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}

		if = {
			limit = {
				has_journal_entry = je_sick_man_education
				has_law_or_variant = law_type:law_state_religion
			}
			add = 1000
		}

		# For Japanese religion Journal Entries
		# If Shinbutsu Bunri is active, but the Taikyo Proclamation has not been made
		if = {
			limit = {
				has_journal_entry = je_shinbutsu_bunri
				country_has_state_religion = rel:mahayana
			}
			add = 1000
		}
	}

 parent = law_freedom_of_conscience
 is_visible = { OR = { c:D3G ?= this c:D3S ?= this } }
 can_enact = { OR = { c:D3G ?= this c:D3S ?= this } }
}
