# I've had enough of Spain winning

movement_mexican_national = {
	category = movement_category_pan_national 

	ideology = ideology_national_liberal_colonial_movement
	character_ideologies = {
		ideology_ethno_nationalist
		ideology_market_liberal
		ideology_liberal_leader
		ideology_republican_leader
		ideology_radical
		ideology_reformer
	}

	can_cause_obstinance = no # for now

	creation_trigger = {
		country_has_primary_culture = cu:spanish
#		is_subject = yes
		c:SC1 ?= this
#		cu:mexican.culture_current_fervor > 10
	}
	creation_weight = {
		value = 100
	}
	on_created = {
		add_movement_enthusiasm_modifier = yes
	}

	disband_trigger = {
		c:MEX ?= this
	}

	on_disbanded = {
	}

	character_support_trigger = {
		OR = {
			has_ideology = ideology:ideology_sovereignist
			has_ideology = ideology:ideology_humanitarian
			culture = { 
				has_discrimination_trait = heritage_latin_american_settler 
			}
#			culture = { cu:mexican }
		}
	}
	character_support_weight = {
		value = 100
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_ethno_nationalist
				}
			}
			multiply = {
				value = 5
			}
		}
		if = {
			limit = {
				OR = {
					has_ideology = ideology:ideology_sovereignist
					has_ideology = ideology:ideology_humanitarian
					has_ideology = ideology:ideology_social_democrat
					has_ideology = ideology:ideology_radical
					has_ideology = ideology:ideology_anarchist
					has_ideology = ideology:ideology_modernizer
					has_ideology = ideology:ideology_utopian
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_vanguardist
				}
			}
			multiply = {
				value = 5
			}
		}
	}

	pop_support_trigger = {
		culture = cu:mexican
	}

	pop_support_factors = {
		movement_support_national_awakening
		movement_support_cultural_fervor
		movement_support_high_urbanization
#		movement_support_latin_settler_heritage 
		movement_support_academics
		movement_support_clerks
		movement_support_shopkeepers
		movement_support_officers
		movement_support_soldiers
		movement_support_nationalism_tech
		movement_support_pan_nationalism_tech
	}

	pop_support_weight = {
		if = {
			limit = {
				exists = scope:political_movement
				scope:political_movement ?= {
					OR = {
						is_being_suppressed = yes
						is_being_bolstered = yes
					}
				}
			}
			if = {
				limit = {
					scope:political_movement = {
						is_being_suppressed = yes
					}
				}
				add = {
					value = suppression_base_value
					multiply = {
						value = -0.5
						add = owner.modifier:country_suppression_attraction_factor
					}
					desc = "concept_suppressed"
				}
			}
			else = {
				add = {
					value = bolstering_base_value
					multiply = {
						value = 0.5
						add = owner.modifier:country_bolster_attraction_factor
					}
					desc = "concept_bolstered"
				}
			}
		}

		if = {
			limit = {
				strata = upper
			}
			if = {
				limit = {
					is_pop_type = capitalists
				}
				add = {
					value = 35
					desc = "POP_CAPITALISTS"
				}
			}
			else = {
				add = {
					value = 30
					desc = "UPPER_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = middle
			}
			if = {
				limit = {
					is_pop_type = academics
				}
				add = {
					value = 35
					desc = "POP_ACADEMICS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = shopkeepers
				}
				add = {
					value = 30
					desc = "POP_SHOPKEEPERS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = officers
				}
				add = {
					value = 30
					desc = "POP_OFFICERS"
				}
			}
			else = {
				add = {
					value = 20
					desc = "MIDDLE_NO_ICON"
				}
			}
		}
		else_if = {
			limit = {
				strata = lower
			}
			if = {
				limit = {
					is_pop_type = clerks
				}
				add = {
					value = 25
					desc = "POP_CLERKS"
				}
			}
			else_if = {
				limit = {
					is_pop_type = soldiers
				}
				add = {
					value = 25
					desc = "POP_SOLDIERS"
				}
			}
			else = {
				add = {
					value = 15
					desc = "LOWER_NO_ICON"
				}
			}
		}

		add = {
			desc = "URBAN_STATE"
			value = 10
			multiply = state.state_urbanization_rate
		}

		if = {
			limit = {
				owner ?= {
					has_technology_researched = pan-nationalism
				}
			}
			multiply = {
				value = 1.5
				desc = "PAN_NATIONALISM_TECH"
			}
		}
		else_if = {
			limit = {
				owner ?= {
					has_technology_researched = nationalism
				}
			}
			multiply = {
				value = 1.25
				desc = "NATIONALISM_TECH"
			}
		}

		multiply = { # Range from x0.775 to 2.275
			desc = "POP_CULTURE_FERVOR"
			value = culture.culture_current_fervor
			subtract = 15
			multiply = 0.015
			add = 1
		}

		if = {
			limit = {
				culture ?= {
					culture_has_national_awakening = yes
				}
			}
			multiply = {
				value = 1.5
				desc = "POP_ACTIVE_NATIONAL_AWAKENING"
			}
		}

		if = { # If a more radical movement exists, it'll poach radicals from this one.
			limit = {
				owner ?= {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_radical
							is_political_movement_type = movement_fascist
						}
					}
				}
			}
			multiply = {
				desc = "POP_RADICAL_MOVEMENT_LOSSES"
				value = 1
				subtract = pop_radical_fraction
			}
		}
	}

	secession = {
		possible = {
			political_movement_support >= 0.05
		}

		weight = {
			value = 1000
		}

		state_weight = {
			value = 1
			if = {
			   limit = {
					NOT = {
						state_region = s:STATE_MEXICO
					}
				}
				multiply = 0
			}
		}

		forced_tags = {
			MEX = {
				trigger = {
					always = yes
				}
				weight = 1000
			}
		}
	}

	additional_radicalism_factors = {
		if = {
			limit = {
				owner ?= {
					is_subject = yes
				}
			}
			add = {
				desc = "INDIA_SUBJECT"
				if = {
					limit = {
						owner ?= {
							OR = {
								is_subject_type = subject_type_colony
								is_subject_type = subject_type_chartered_company
							}
						}
					}
					value = 0.25
				}
				if = {
					limit = {
						owner ?= {
							is_subject_type = subject_type_dominion
						}
					}
					value = 0.10
				}
			}
		}
		if = {
			limit = {
				owner ?= {
					is_subject = yes
				}
			}
			add = {
				desc = "LIBERTY_DESIRE"
				value = owner.liberty_desire
				divide = 500
			}
		}

		if = {
			limit = {
				owner = {
					any_neighbouring_state = {
						owner = {
							NOT = { THIS = ROOT.owner }
							any_political_movement = {
								is_political_movement_type = movement_cultural_minority
								is_insurrectionary = yes
							}
						}
					}
				}
			}
			add = {
				desc = "NEIGHBOURING_COUNTRY_WITH_REVOLUTION"
				value = 0.15
			}
		}
	}
}
