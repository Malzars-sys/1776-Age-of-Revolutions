# R4: article 1 is a native weekly transfer, article 2 is the US debt promise.
aor1776_support_money = {
 kind = directed cost = 0
 icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/bankroll_treaties.dds"
 flags = { is_money_transfer }
 maintenance_paid_by = target_country
 usage_limit = once_per_treaty
 required_inputs = { quantity }
 visible = { OR = { c:FRA ?= this c:USA ?= this } }
 # Only the payer must be solvent; aid remains possible for a debtor in default.
 possible = { c:FRA = { in_default = no } }
 can_ratify = {
  scope:source_country = { c:FRA ?= this in_default = no }
  scope:target_country = { c:USA ?= this }
 }
 quantity_min_value = { value = 30000 }
 quantity_max_value = { value = 30000 }
 requirement_to_maintain = {
  trigger = {
   aor1776_support_contract_parties_valid = yes
   # This is also checked on article options before ratification. An options
   # scope cannot follow the runtime article -> treaty link (1.13 log error).
   c:USA = { OR = { has_variable = aor1776_support_offer has_variable = aor1776_support_active } }
  }
 }
 non_fulfillment = {
  consequences = freeze
  conditions = { weekly = { scope:article = { source_country = { in_default = yes } } } }
 }
 can_withdraw = { c:USA = { NOT = { has_variable = aor1776_debt_binding_lock } } }
}

# Source is the debtor making the promise; target is its creditor.
aor1776_deferred_debt_article = {
 kind = directed cost = 0
 icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/aor1776_deferred_debt.dds"
 maintenance_paid_by = source_country
 usage_limit = once_per_side
 visible = { NOT = { is_country_type = decentralized } }
 possible = { NOT = { has_war_with = scope:other_country } }
 can_ratify = {
  scope:source_country = {
   OR = {
    AND = { c:USA ?= this has_variable = aor1776_support_offer scope:target_country = { c:FRA ?= this } }
    AND = { has_variable = aor1776_debt_creditor var:aor1776_debt_creditor = scope:target_country }
   }
  }
 }
 requirement_to_maintain = {
  trigger = {
   scope:source_country = {
    OR = {
     has_variable = aor1776_support_offer
     AND = { has_variable = aor1776_debt_creditor var:aor1776_debt_creditor = scope:target_country }
    }
   }
  }
 }
 can_withdraw = { scope:source_country = { NOT = { has_variable = aor1776_debt_binding_lock } } }
}

aor1776_support_supplies = {
 kind = directed cost = 0
 icon = "gfx/interface/icons/diplomatic_treaties_articles_icons/goods_transfer_treaties.dds"
 flags = { is_goods_transfer }
 maintenance_paid_by = source_country
 required_inputs = { quantity goods }
 visible = { OR = { c:FRA ?= this c:USA ?= this } }
 possible = {
  OR = { c:FRA ?= this c:USA ?= this }
  is_market_reachable_for_trade = scope:other_country.market
 }
 can_ratify = {
  scope:source_country = { c:FRA ?= this }
  scope:target_country = { c:USA ?= this }
 }
 requirement_to_maintain = {
  trigger = {
   aor1776_support_contract_parties_valid = yes
   c:USA = { OR = { has_variable = aor1776_support_offer has_variable = aor1776_support_active } }
  }
 }
 non_fulfillment = {
  consequences = freeze
  conditions = {
   weekly = {
    scope:article = {
     OR = {
      source_country = { supply_network_strength < 0.5 }
      NOT = { source_country = { is_market_reachable_for_trade = scope:article.target_country.market } }
      "scope:article.input_market_goods(scope:article.source_country)" = { treaty_exports > market_export_cap }
     }
    }
   }
  }
 }
 goods_valid_trigger = { scope:input = { OR = { this = g:small_arms this = g:artillery } } }
 quantity_min_value = { value = 1 }
 quantity_max_value = { value = 15 }
}
