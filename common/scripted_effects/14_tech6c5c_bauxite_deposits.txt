tech6c_reveal_bauxite_deposits = {
	if = {
		limit = {
			NOT = { has_global_variable = tech6c_bauxite_deposits_revealed }
		}
		set_global_variable = tech6c_bauxite_deposits_revealed
		tech6c_bauxite_nano_add_deposits = yes
		tech6c_bauxite_small_add_deposits = yes

		tech6c_bauxite_big_add_deposits = yes
		tech6c_bauxite_huge_add_deposits = yes
	}
}
