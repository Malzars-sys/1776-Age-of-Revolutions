je_gabelle = {
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	group = je_group_historical_content

	is_shown_when_inactive = {
		c:FRA ?= this
		NOT = { has_variable = lagab_gabelle_reformed }
		has_consumption_tax = g:salt
	}

	possible = {
		c:FRA ?= this
		has_consumption_tax = g:salt
	}

	# This active JE is destroyed when the tax disappears. The canonical
	# monthly country controller creates a fresh instance after tax restoration.
	modifiers_while_active = {
		gabelle_income_active
	}

	invalid = {
		NOT = { has_consumption_tax = g:salt }
	}

	custom_completion_header = "je_gabelle_complete_header"
	complete = {
		custom_tooltip = {
			text = "je_gabelle_reform_goal"
			has_variable = lagab_gabelle_reformed
		}
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
	
}
