                   type = bg_light_industry

                 }
                    state_has_building_group_levels = {
                        type = bg_heavy_industry
                        value >= 8
                    }
                    state_has_building_group_levels = {
                        type = bg_military_industry
                        value >= 8
                    }
                }

		}

    on_complete = {

                    multiplier = 0.5
                }

                every_scope_state = {
                    limit = {
                        OR = {
                            state_has_building_group_levels = {
                                type = bg_light_industry
                                value >= 7
                            }
                            state_has_building_group_levels = {
                                type = bg_heavy_industry
                                value >= 4
                            }

                                type = bg_military_industry
                                value >= 4
                            }

                        name = mod_polish_miracle_state

je_plc_reform_farming = { # Farming Revolution
    icon = "gfx/interface/icons/invention_icons/threshing_machine.dds"

    group = je_group_historical_content

    possible = {
        # triggered by history
    }

    immediate = { 

    }

    complete = {
		NOT = { has_law = law_type:law_serfdom }
		
		any_scope_state = {
            any_scope_building = {
                filter = {
                    is_building_group = bg_staple_crops
                    state = {
                        state_region = {
                            is_homeland = cu:polish
                        }
                    }
                }
                OR = {
                    has_active_production_method = pm_soil_enriching_farming
                    hidden_trigger = {
                        has_active_production_method = pm_soil_enriching_farming_building_rice_farm
                    }
                }
                percent >= 0.8
            }
            any_scope_building = {
                filter = {
                    is_building_group = bg_staple_crops
                    state = {
                        state_region = {
                            is_homeland = cu:polish
                        }
                    }
                }
                OR = {
                    has_active_production_method = pm_tools
                    hidden_trigger = {
                        has_active_production_method = pm_tools_building_rice_farm
                    }
                }
                percent >= 0.8
            }
        }
        country_has_building_group_levels = {
                type = bg_staple_crops
                value  >= 15
            }
    }

    on_complete = {
		add_modifier = {
			name = modifier_polish_farming_boom
			days = even_longer_modifier_time
			multiplier = 0.5
		}		
    }

    weight = 100
    should_be_pinned_by_default_uninvolved_or_context = yes
}
