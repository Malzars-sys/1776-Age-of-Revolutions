je_romanticism = {

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	group = je_group_technology

	is_shown_when_inactive = {
		has_technology_researched = institutionalized_scientific_exchange
		NOT = { has_technology_researched = realism }
		NOT = { has_variable = completed_romanticism_je }
		NOT = { is_country_type = unrecognized }
	}

	possible = {
		has_technology_researched = romanticism
		NOT = { has_technology_researched = realism }
		any_scope_state = {
			has_building = building_urban_center
		}
	}

	immediate = {
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_building = {
				is_building_type = building_art_academy
				has_active_production_method = pm_traditional_art
				occupancy >= 0.9
				level >= 2
			}
		}
		effect = {
			trigger_event = { id = society_tech_events.302 }
			set_variable = {
				name = completed_romanticism_je
			}
		}
	}


	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_technology_researched = realism
		}
		effect = {}
	}

	timeout = 3650
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {}
	}
}

je_camera = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_technology

	is_shown_when_inactive = {
		NOT = { has_variable = camera_je_done }
		has_technology_researched = realism
		NOT = { has_technology_researched = film }
		NOT = { is_country_type = unrecognized }
	}

	possible = {
		has_technology_researched = camera
		any_scope_building = {
			is_building_type = building_art_academy
			level >= 3
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_building = {
				is_building_type = building_art_academy
				level >= 5
				has_active_production_method = pm_photographic_art
			}
			any_scope_building = {
				filter = {
					is_building_type = building_art_academy
				}
				percent > 0.75
				has_active_production_method = pm_photographic_art
				occupancy > 0.75
			}
		}
		effect = {
			set_variable = camera_je_done
			trigger_event = { id = camera_film_pm_events.1 }
		}
	}


	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			has_technology_researched = film
		}
		effect = {}
	}

	timeout = 3650
	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {}
	}
}
