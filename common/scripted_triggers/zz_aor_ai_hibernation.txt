aor_ai_running = {
    NOT = { has_global_variable = aor_ai_suspended }
}

aor_ai_recovery_required = {
    aor_ai_registered = yes
    OR = {
        has_variable = aor_ai_restore_target
        AND = {
            aor_ai_running = yes
            aor_ai_active_requirement = yes
        }
    }
}

aor_ai_wake_required = {
    OR = {
        has_variable = aor_ai_restore_target
        AND = {
            aor_ai_running = yes
            OR = {
                aor_ai_active_requirement = yes
                AND = {
                    NOT = { has_variable = aor_ai_recovery_window }
                    has_variable = aor_ai_watched
                }
            }
        }
    }
}

aor_ai_registered = {
    has_global_variable = aor_ai_initialized
    has_variable = aor_ai_member
    is_country_alive = yes
}

aor_ai_country_is_eligible_for_hibernation = {
    aor_ai_registered = yes
    NOT = { has_variable = aor_ai_promoted }
}

# A start-of-campaign integration may commit a lock once the start hook has armed the system.
aor_ai_can_record_lock = {
    aor_ai_running = yes
    is_country_alive = yes
    OR = {
        aor_ai_country_is_eligible_for_hibernation = yes
        AND = {
            has_global_variable = aor_ai_armed
            NOT = { has_global_variable = aor_ai_initialized }
        }
    }
}

# Last dispatch attempt, not a read of the native controller.
aor_ai_country_ai_is_hibernating = {
    aor_ai_country_is_eligible_for_hibernation = yes
    has_variable = aor_ai_sleep_sent
    NOT = { has_variable = aor_ai_wake_pending }
    is_player = no
}

aor_ai_player_conflict = {
    OR = {
        any_enemy_in_diplo_play = { is_player = yes }
        any_cobelligerent_in_diplo_play = { is_player = yes }
    }
}

aor_ai_own_conflict = {
    OR = {
        is_at_war = yes
        is_diplomatic_play_committed_participant = yes
    }
}

aor_ai_active_requirement = {
    OR = {
        is_player = yes
        overlord ?= { is_player = yes }
        has_variable = aor_ai_promoted
        has_variable = aor_ai_conflict_grace
        AND = {
            has_variable_map = aor_ai_gameplay_locks
            variable_map_size = { name = aor_ai_gameplay_locks value > 0 }
        }
        aor_ai_player_conflict = yes
        AND = {
            aor_ai_own_conflict = yes
            OR = {
                has_variable = aor_ai_watched
                NOT = { has_variable = aor_ai_sleep_sent }
            }
        }
    }
}

aor_ai_can_dispatch_sleep = {
    aor_ai_running = yes
    aor_ai_country_is_eligible_for_hibernation = yes
    is_player = no
    NOT = { has_variable = aor_ai_watched }
    NOT = { has_variable = aor_ai_wake_pending }
    NOT = { aor_ai_active_requirement = yes }
    NOT = { aor_ai_own_conflict = yes }
}

aor_ai_wake_dispatchable = {
    aor_ai_registered = yes
    aor_ai_wake_required = yes
    has_variable = aor_ai_pending
    has_variable = aor_ai_wake_pending
    NOT = { has_variable = aor_ai_packet_sleep }
}

aor_ai_sleep_dispatchable = {
    aor_ai_registered = yes
    has_variable = aor_ai_pending
    aor_ai_can_dispatch_sleep = yes
}
