	name = "je_french_revolution_skip"
	desc = "je_french_revolution_skip_desc"

	visible = {
		c:FRA ?= {
			is_player = yes
		}
	}

	cooldown = { days = 7 }

	possible = {
		is_at_war = no
	}

	effect = {
		activate_law = law_type:law_tenant_farmers
		activate_law = law_type:law_freedom_of_conscience

			add_progress = { value = 120 name = je_estates_reform_bar }
		}
		trigger_event = { 
			id = french_revolution_mod_events.19 popup = yes 
		}
	}
}
