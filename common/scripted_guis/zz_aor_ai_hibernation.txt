        is_player = yes

aor_ai_relay_can_start_sleep = {
    scope = country
    is_valid = {
        is_player = yes
        has_global_variable = aor_ai_initialized
        aor_ai_running = yes
        has_global_variable = aor_ai_slot_0
        OR = { NOT = { any_in_global_list = { variable = aor_ai_pending_countries aor_ai_wake_dispatchable = yes } } AND = { NOT = { any_in_global_list = { variable = aor_ai_pending_countries aor_ai_wake_dispatchable = yes NOT = { has_variable = aor_ai_recovery_window } } } global_var:aor_ai_wakes_since_sleep >= 4 } }
        NOT = { has_global_variable = aor_ai_wake_slot }
    }
}

        if = { limit = { NOT = { has_global_variable = aor_ai_wakes_since_sleep } } set_global_variable = { name = aor_ai_wakes_since_sleep value = 0 } }
        if = { limit = { global_var:aor_ai_wakes_since_sleep < 4 } change_global_variable = { name = aor_ai_wakes_since_sleep add = 1 } }
        global_var:aor_ai_wake_slot ?= { aor_ai_record_wake_attempt = yes }
        remove_global_variable = aor_ai_wake_slot
    }
}
