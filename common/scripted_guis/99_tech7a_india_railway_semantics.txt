REPLACE_OR_CREATE:je_india_railway_delhi_calcutta_sgui = {
    scope = country

    effect = {
        if = {
            limit = {
                NOT = {
                    any_state_region_in_delhi_calcutta_line = {
                        state_has_active_rail_network = yes
                        count = all
                    }
                }
            }
            every_state_region_in_delhi_calcutta_line = {
                if = {
                    limit = {
                        state_has_active_rail_network = yes
                    }
                    save_temporary_scope_as = state_region_to_display
                    custom_tooltip_no_bullet = "INDIAN_RAILWAY_ENTRY" # for each print a line
                }
                else = {
                    save_temporary_scope_as = state_region_to_display
                    custom_tooltip_no_bullet = "INDIAN_RAILWAY_ENTRY_NO" # for each print a line
                }
            }
        }
    }
}
