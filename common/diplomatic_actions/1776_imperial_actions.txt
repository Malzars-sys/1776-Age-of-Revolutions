  is_diplomatic_play_committed_participant = no
  NOT = { has_war_with = scope:target_country }
  scope:target_country = { is_diplomatic_play_committed_participant = no }

  scope:second_state = { aor1776_hre_restitution_state = yes NOT = { aor1776_hre_restitution_cooldown = yes } }

  owner = scope:target_country

  NOT = { aor1776_hre_restitution_cooldown = yes }

  aor1776_hre_return_requested_provinces = yes
  trigger_event = { id = aor1776_hre.32 popup = yes }
 }
}

  will_propose = { aor1776_hre_emperor = yes }

  propose_score = { value = 100 }

 value = 0 add = { value = 1000 desc = aor1776_hre_accept_imperial_duty }
} else = { value = 0
add = { value = -25 desc = aor1776_hre_accept_base }
add = { value = "relations(scope:actor)" multiply = 0.5 desc = aor1776_hre_accept_relations }
add = { value = "scope:actor.ai_army_comparison(root)" multiply = 2 min = -50 max = 50 desc = aor1776_hre_accept_army }
add = { value = aor1776_hre_foreign_allied_strength divide = { value = scope:actor.aor1776_hre_imperial_available_strength min = 1 } multiply = -40 min = -60 max = 0 desc = aor1776_hre_accept_allies }
if = { limit = { scope:actor.power_bloc ?= { has_identity = identity:identity_aor1776_imperial_federation } }
 add = { value = scope:actor.power_bloc.current_cohesion_number subtract = 50 multiply = 0.75 desc = aor1776_hre_accept_cohesion }
}
if = { limit = { owes_obligation_to = scope:actor } add = { value = 60 desc = aor1776_hre_accept_obligation } }
 } }
}
