  NOT = { is_country_type = decentralized }

  scope:target_country = {
   set_variable = { name = aor1776_debt_creditor value = root }

   set_variable = { name = aor1776_debt_interest value = 0 }
   aor1776_debt_generic_advance_effect = yes
  }
 }

  cost = 0
  is_two_sided_pact = no
  actor_can_break = { always = no }
  target_can_break = { always = no }

  requirement_to_maintain = {
   trigger = {
    NOT = { is_country_type = decentralized }
    scope:target_country = { NOT = { is_country_type = decentralized } }
   }
  }
 }

}

  }
