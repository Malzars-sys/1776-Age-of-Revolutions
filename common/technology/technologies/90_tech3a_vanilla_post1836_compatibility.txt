	}

		if = {
			limit = {
				NOT = {
					any_political_movement = {
						is_political_movement_type = movement_modernizer
					}
				}
				OR = {
					AND = {
						OR = {
							is_in_geographic_region = geographic_region_south_america
							is_in_geographic_region = geographic_region_central_america_old
						}
						country_rank <= rank_value:major_power
					}
					country_is_colonial_or_company = yes
					is_country_type = unrecognized
				}
				NAND = {
					has_law_or_variant = law_type:law_protectionism
					OR = {
						has_law_or_variant = law_type:law_interventionism
						has_law_or_variant = law_type:law_command_economy
						has_law_or_variant = law_type:law_laissez_faire
					}
					has_law_or_variant = law_type:law_compulsory_primary_school
				}
			}
			create_political_movement = { type = movement_modernizer }
		}
	}

	ai_weight = {
		value = 1.5 # Important in general
	}

					limit = {
						has_core_ideology = ideology:ideology_carlist_movement

					limit = {
						has_ideology = ideology:ideology_carlist_ig

					remove_ideology = ideology_carlist_ig
					add_ideology = ideology_carlist_ig_2
				}
			}
		}
		if = {
			limit = {
				country_has_primary_culture = cu:portuguese
			}
			custom_tooltip = {
				text = miguelist_amended_tt
				every_character = {

						has_ideology = ideology:ideology_miguelist
