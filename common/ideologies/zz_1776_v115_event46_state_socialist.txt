# Canonical 1.15 providers retained by the annual event lists and their dependencies.
ideology_state_socialist = { # Average Progressiveness - 65.62
	icon = "gfx/interface/icons/ideology_icons/ideology_leader/ideology_leader_socialist.dds"

	character_ideology = yes

	lawgroup_distribution_of_power = {
		law_universal_suffrage = approve
		law_census_voting = neutral
		law_wealth_voting = disapprove
		law_landed_voting = disapprove
		law_anarchy = disapprove
		law_technocracy = disapprove
		law_oligarchy = strongly_disapprove
		law_autocracy = strongly_disapprove
		law_single_party_state = neutral
		law_labour_suffrage = strongly_disapprove
	}

	lawgroup_caste_hegemony = {
		law_hindu_caste_enforced = strongly_disapprove
		law_hindu_caste_codified = disapprove
		law_hindu_caste_not_enforced = neutral
		law_affirmative_action = approve
	}

	lawgroup_edo_social_system = {
		law_strict_edo_system = strongly_disapprove
		law_intermediate_edo_system = disapprove
		law_lax_edo_system = approve
	}

	lawgroup_economic_system = {
		law_command_economy = strongly_approve
		law_cooperative_ownership = approve
		law_interventionism = neutral
		law_agrarianism = disapprove
		law_traditionalism = disapprove
		law_laissez_faire = strongly_disapprove
		law_industry_banned = strongly_disapprove
		law_extraction_economy = strongly_disapprove
	}

	lawgroup_taxation = {
		law_consumption_based_taxation = disapprove
		law_land_based_taxation = disapprove
		law_per_capita_based_taxation = neutral
		law_proportional_taxation = approve
		law_graduated_taxation = strongly_approve
	}

	lawgroup_labor_rights = {
		law_worker_protections = strongly_approve
		law_regulatory_bodies = approve
		law_no_workers_rights = disapprove
	}

	lawgroup_welfare = {
		law_poor_laws = disapprove
		law_no_social_security = strongly_disapprove
		law_wage_subsidies = approve
		law_old_age_pension = strongly_approve
	}

	lawgroup_labour_associations = {
		law_combination_acts = strongly_disapprove
		law_guild_system = disapprove
		law_anti_strike_laws = neutral
		law_corporatized_unions = neutral
		law_right_to_associate = approve
		law_factory_councils = strongly_disapprove
	}

	country_trigger = {
		has_technology_researched = socialism
	}

	interest_group_leader_trigger = {
		ideology_state_socialist_valid_trigger = yes
	}

	non_interest_group_leader_trigger = {
		# Agitator ideologies should disapprove of a current law
		NAND = {
			has_role_of_type = agitator
			owner ?= {
				has_law_or_variant = law_type:law_universal_suffrage
				has_law_or_variant = law_type:law_graduated_taxation
				has_law_or_variant = law_type:law_old_age_pension
				has_law_or_variant = law_type:law_worker_protections
			}
		}
		interest_group ?= {
			ideology_state_socialist_valid_trigger = yes
		}
	}

	interest_group_leader_weight = {
		value = {
			add = 100
			desc = "base_value"
		}

		# More likely if issues are relevant
		if = {
			limit = {
				owner.ideology_state_socialist_disfavoured_laws > 0
			}
			add = {
				value = owner.ideology_state_socialist_disfavoured_laws
				multiply = 9 # Caste laws don't count by default
				desc = "DISFAVORED_LAWS"
			}
		}

		# very angry IG's more likely to turn to more radical socialism
		if = {
			limit = {
				NOT = { is_interest_group_type = ig_devout } # can't be radical socialists
				ig_approval <= define:NPolitics|APPROVAL_THRESHOLD_UNHAPPY
			}
			add = {
				value = -75
				desc = "IG_APPROVAL_BELOW_UNHAPPY"
			}
		}
		# more likely for content devout in council republic
		if = {
			limit = {
				is_interest_group_type = ig_devout
				ig_approval > 0
				owner ?= {
					has_law_or_variant = law_type:law_council_republic
				}
			}
			add = {
				value = 75
				desc = "CONTENT_IN_CR"
			}
		}
		# more likely for content Trade Unions outside of council republic
		if = {
			limit = {
				is_interest_group_type = ig_trade_unions
				ig_approval > 0
				owner ?= {
					NOT = { has_law_or_variant = law_type:law_council_republic }
				}
			}
			add = {
				value = 75
				desc = "CONTENT_TU"
			}
		}

		# liberal and/or radical leaders have a higher spawn chance if the Dreyfusards won.
		if = {
			limit = {
				has_variable = dreyfusard_victory
			}
			add = {
				value = 50
				desc = "DREYFUSARD_VICTORY"
			}
		}
		# liberal and/or radical leaders have a higher spawn chance if you win the Two Spains as the liberals.
		if = {
			limit = {
				has_variable = spa_two_spains_liberal_victory
			}
			add = {
				value = 50
				desc = "SPA_LIBERAL_VICTORY"
			}
		}
		# liberal and/or radical leaders have a higher spawn chance if you succeed in Portugal's Second Liberalism.
		if = {
			limit = {
				has_variable = por_second_liberalism_victory
			}
			add = {
				value = 50
				desc = "POR_LIBERAL_VICTORY"
			}
		}
		# liberal and/or radical leaders have a higher spawn chance if the Hungarian Revolution was successful.
		if = {
			limit = {
				has_variable = hungarian_revolution_increased_liberalism
			}
			add = {
				value = 50
				desc = "HUNGARIAN_REVOLUTION"
			}
		}
		# Pedro JE victory
		if = {
			limit = {
				owner ?= {
					has_variable = brz_enduring_empire
				}
			}
			add = {
				value = 50
				desc = "BRZ_ENDURING_EMPIRE"
			}
		}
		# extremely unlikely for Trade Unions in council republic
		if = {
			limit = {
				is_interest_group_type = ig_trade_unions
				owner ?= {
					has_law_or_variant = law_type:law_council_republic
				}
			}
			multiply = {
				value = 0.01
				desc = "COUNCIL_REPUBLIC_LAW"
			}
		}
		# National Liberation Alliance
		if = {
			limit = {
				owner ?= {
					has_variable = national_liberation_alliance_variable
					has_journal_entry = je_new_republic
				}
			}
			multiply = {
				value = 1.20
				desc = "NATIONAL_LIBERATION_ALLIANCE"
			}
		}
		if = {
			limit = {
				OR = {
					has_variable = communist_ig_var
					has_variable = chose_vanguardism
					has_variable = chose_anarchism
					has_variable = chose_communism
					has_variable = chose_ethno_nationalism
					has_variable = chose_conservatism
					has_variable = chose_fascism
				}
			}
			multiply = {
				value = 0.1
				desc = "CHOSE_A_DIFFERENT_PATH"
			}
		}
		multiply = {
			value = 1
			add = modifier:interest_group_ideology_state_socialist_leader_chance_mult
			desc = "EVENTS_AND_OTHER_EFFECTS"
		}
	}

	non_interest_group_leader_weight = {
		value = 100

		# More likely if issues are relevant
		if = {
			limit = {
				owner.ideology_state_socialist_disfavoured_laws > 0
			}
			add = {
				value = owner.ideology_state_socialist_disfavoured_laws
				multiply = 9 # Caste laws don't count by default
			}
		}

		# very angry IG's more likely to turn to more radical socialism
		if = {
			limit = {
				NOT = { is_interest_group_type = ig_devout } # can't be radical socialists
				interest_group.ig_approval <= define:NPolitics|APPROVAL_THRESHOLD_UNHAPPY
			}
			add = -75
		}
		# more likely for content devout in council republic
		if = {
			limit = {
				is_interest_group_type = ig_devout
				interest_group.ig_approval > 0
				owner ?= {
					has_law_or_variant = law_type:law_council_republic
				}
			}
			add = 75
		}
		# more likely for content Trade Unions outside of council republic
		if = {
			limit = {
				is_interest_group_type = ig_trade_unions
				interest_group.ig_approval > 0
				owner ?= {
					NOT = { has_law_or_variant = law_type:law_council_republic }
				}
			}
			add = 75
		}
		# liberal and/or radical leaders have a higher spawn chance if you win the Two Spains as the liberals and choose to have woke commanders.
		if = {
			limit = {
				OR = {
					has_role_of_type = general
					has_role_of_type = admiral
				}
				has_variable = spa_liberal_military_var
			}
			add = 50
		}
		# extremely unlikely for Trade Unions in council republic
		if = {
			limit = {
				is_interest_group_type = ig_trade_unions
				owner ?= {
					has_law_or_variant = law_type:law_council_republic
				}
			}
			multiply = 0.01
		}
		if = {
			limit = {
				interest_group ?= {
					OR = {
						has_variable = communist_ig_var
						has_variable = chose_vanguardism
						has_variable = chose_anarchism
						has_variable = chose_communism
						has_variable = chose_ethno_nationalism
						has_variable = chose_conservatism
						has_variable = chose_fascism
					}
				}
			}
			multiply = 0.1
		}
		multiply = {
			value = 1
			add = interest_group.modifier:interest_group_ideology_state_socialist_leader_chance_mult
			desc = "EVENTS_AND_OTHER_EFFECTS"
		}
	}
}
